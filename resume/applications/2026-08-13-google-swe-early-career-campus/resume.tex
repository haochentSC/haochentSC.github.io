%  3. \resumeProjectHeading takes a third argument, the tech-stack line. It is typeset
%     inside the same \item as the heading so it collects no list glue. Do NOT give the
%     stack its own \item: a stack list is not an accomplishment, and a bare \item[] also
%     drags in \itemsep + \parsep, which is what pushed this resume onto a second page.
%  4. List spacing is set explicitly via enumitem instead of the stock template's stack of

\newcommand{\resumeItem}[1]{\item\small{#1}}

% The trailing \vspace values here and in \resumeProjectHeading cancel the following list's
% \topsep. They are tuned against \topsep=2pt below, NOT against the stock 9pt default:
% restore the stock -7pt and the stack line lands on top of the first bullet.
\newcommand{\resumeSubheading}[4]{
  \item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-2pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}

% #1 project name (with optional link), #2 dates, #3 tech stack.
% The stack rides inside this \item on its own line: no bullet, no list glue, and it wraps
% like ordinary text instead of overflowing the way a tabular row would.
\newcommand{\resumeProjectHeading}[3]{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & \small#2 \\
    \end{tabular*}
    \par\vspace{-4pt}
    \noindent{\footnotesize #3}
    \par\vspace{-2pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

% Explicit list spacing. The stock template leaves \topsep/\parsep/\itemsep at their 11pt
% defaults (~9pt each) and claws the space back with negative \vspace; that is engine- and
% font-sensitive. Setting them outright is the single knob for overall density: raise
% itemsep below to loosen the page, lower it to tighten.
\newcommand{\resumeSubHeadingListStart}{%
  \begin{itemize}[leftmargin=0.15in, label={}, topsep=2pt, partopsep=0pt, parsep=0pt, itemsep=4pt]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{%
  \begin{itemize}[leftmargin=0.25in, topsep=2pt, partopsep=0pt, parsep=0pt, itemsep=1pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-3pt}}
